            constructor() {

                this.isPlaying = false;

            }


            init() {

                if (!this.ctx) {

                    const AudioCtx = window.AudioContext || window.webkitAudioContext;

                    this.ctx = new AudioCtx();

                }

            }


            stop() {

                this.activeNodes.forEach(node => {

                    try { node.stop(); } catch(e){}

                    try { node.disconnect(); } catch(e){}

                });

                this.activeNodes = [];

                this.isPlaying = false;

                updateAudioUI();

            }


            play(track = this.currentTrack) {

                this.init();

                this.stop();

                this.currentTrack = track;

                this.isPlaying = true;


                if (this.ctx.state === 'suspended') {

                    this.ctx.resume();

                }


                const masterGain = this.ctx.createGain();

                masterGain.gain.setValueAtTime(this.volume, this.ctx.currentTime);

                masterGain.connect(this.ctx.destination);


                if (track === 'lullaby') {

                    const notes = [261.63, 329.63, 392.00, 523.25, 392.00, 329.63];

                    let step = 0;

                    const interval = setInterval(() => {

                        if (!this.isPlaying) return clearInterval(interval);

                        const osc = this.ctx.createOscillator();

                        const gain = this.ctx.createGain();

                        osc.type = 'sine';

                        osc.frequency.setValueAtTime(notes[step % notes.length], this.ctx.currentTime);

                        gain.gain.setValueAtTime(0.15, this.ctx.currentTime);

                        gain.gain.exponentialRampToValueAtTime(0.0001, this.ctx.currentTime + 2.5);

                        osc.connect(gain);

                        gain.connect(masterGain);

                        osc.start();

                        osc.stop(this.ctx.currentTime + 2.6);

                        step++;

                    }, 1400);

                } else if (track === 'waves') {

                    const bufferSize = this.ctx.sampleRate * 2;

                    const data = buffer.getChannelData(0);

                    let b0=0, b1=0, b2=0, b3=0, b4=0, b5=0, b6=0;

                    for (let i = 0; i < bufferSize; i++) {

                        let white = Math.random() * 2 - 1;

                        b0 = 0.99886 * b0 + white * 0.0555179;

                        b1 = 0.99332 * b1 + white * 0.0750759;

                        b2 = 0.96900 * b2 + white * 0.1538520;

                        data[i] = b0 + b1 + b2 + b3 + b4 + b5 + b6 + white * 0.5362;

                        data[i] *= 0.11;

                        b6 = white * 0.115926;

                    }


                    const noise = this.ctx.createBufferSource();

                    noise.buffer = buffer;

                    noise.loop = true;

                    filter.type = 'lowpass';

                    filter.frequency.value = 300;


                    const lfo = this.ctx.createOscillator();

                    lfo.frequency.value = 0.1;

                    const lfoGain = this.ctx.createGain();

                    lfoGain.gain.value = 200;


                    lfo.connect(lfoGain);

                    lfoGain.connect(filter.frequency);


                    noise.connect(filter);

                    filter.connect(masterGain);


                    lfo.start();

                    noise.start();

                    this.activeNodes.push(noise, lfo);

                } else if (track === 'crickets' || track === 'whitenoise') {

                    const bufferSize = this.ctx.sampleRate * 2;

                    filter.type = track === 'crickets' ? 'bandpass' : 'lowpass';

                                ${savedPage !== undefined ? `<span class="text-[10px] text-gray-500"><i class="fa-solid fa-book-open"></i> Pg ${savedPage + 1}</span>` : ''}

                                <button class="btn-card-bookmark text-xs ${isBookmarked ? 'text-amber-400' : 'text-gray-600 hover:text-gray-400'}" data-id="${story.id}">

                                    <i class="fa-solid fa-bookmark"></i>

                                </button>

                            </div>

                        </div>

                        <h3 class="font-display text-lg text-kindle-cream group-hover:text-kindle-amber transition line-clamp-1">${story.title}</h3>

                        <p class="text-xs text-gray-500 font-sans mt-1 italic">${story.origin}</p>

                    </div>

                    <div class="mt-4 pt-3 border-t border-kindle-border/40 flex items-center justify-between text-xs text-gray-400 font-sans">

                        <span class="text-kindle-amber/90 font-medium">${story.content.length} Full Pages</span>

                        <span class="text-kindle-amber opacity-0 group-hover:opacity-100 transition flex items-center gap-1">Read <i class="fa-solid fa-arrow-right text-[10px]"></i></span>

                    </div>

                `;


                card.addEventListener('click', (e) => {

                    if (e.target.closest('.btn-card-bookmark')) {

                        toggleBookmark(story.id);

                        e.stopPropagation();

                        return;

                    }

                    const pageToOpen = APP_STATE.readingHistory[story.id] || 0;

                    openStory(story, pageToOpen);

                });


                el.storyGrid.appendChild(card);

            });

        }


        function openStory(story, pageIndex = 0) {

            if (!checkTrialStatus()) return;

            APP_STATE.currentStory = story;

            APP_STATE.currentPage = pageIndex;


            el.readerTitle.textContent = story.title;

            el.readerAgeBadge.textContent = `Age ${story.age}`;

            el.readerOrigin.textContent = story.origin;


            updateReaderContent();

            updateBookmarkBtnUI();


            el.viewDashboard.classList.add('hidden');

            el.viewReader.classList.remove('hidden');


            window.scrollTo({ top: 0, behavior: 'smooth' });

        }


        function updateReaderContent() {

            const story = APP_STATE.currentStory;

            if (!story) return;


            const totalPages = story.content.length;

            const curPage = APP_STATE.currentPage;


            el.readerText.classList.remove('page-active');

            el.readerText.classList.add('page-enter');


            setTimeout(() => {

                el.readerText.textContent = story.content[curPage];

                el.readerText.classList.remove('page-enter');

                el.readerText.classList.add('page-active');

            }, 120);


            el.pageIndicator.textContent = `Page ${curPage + 1} of ${totalPages}`;

            const progressPercent = ((curPage + 1) / totalPages) * 100;

            el.readProgressFill.style.width = `${progressPercent}%`;


            el.btnPrevPage.disabled = curPage === 0;

            el.btnNextPage.disabled = curPage === totalPages - 1;


            APP_STATE.readingHistory[story.id] = curPage;

            localStorage.setItem('nocturne_progress', JSON.stringify(APP_STATE.readingHistory));

        }


        function applySettings() {

            const s = APP_STATE.settings;

            el.readerText.style.fontSize = `${s.fontSize}px`;

            el.readerText.style.color = s.color;

            

            el.readerText.className = `page-content page-active w-full text-justify leading-relaxed tracking-wide transition-all ${s.font}`;

            

            el.dimmerOverlay.style.opacity = s.dimmer / 100;

            el.dimmerVal.textContent = `${s.dimmer}%`;

            el.fontSizeVal.textContent = `${s.fontSize}px`;


            localStorage.setItem('nocturne_settings', JSON.stringify(s));

        }


        function toggleBookmark(storyId) {

            const idx = APP_STATE.bookmarks.indexOf(storyId);

            if (idx > -1) {

                APP_STATE.bookmarks.splice(idx, 1);

            } else {

                APP_STATE.bookmarks.push(storyId);

            }

            localStorage.setItem('nocturne_bookmarks', JSON.stringify(APP_STATE.bookmarks));

            updateBookmarkBtnUI();

            renderDashboard();

        }


        function updateBookmarkBtnUI() {

            if (!APP_STATE.currentStory) return;

            const isBookmarked = APP_STATE.bookmarks.includes(APP_STATE.currentStory.id);

            if (isBookmarked) {

                el.btnBookmark.classList.remove('text-gray-500');

                el.btnBookmark.classList.add('text-amber-400');

            } else {

                el.btnBookmark.classList.remove('text-amber-400');

                el.btnBookmark.classList.add('text-gray-500');

            }

        }


        function updateAudioUI() {

            if (audioEngine.isPlaying) {

                el.audioIcon.className = "fa-solid fa-volume-high text-kindle-amber audio-playing";

                el.audioStatusText.textContent = "Playing";

            } else {

                el.audioIcon.className = "fa-solid fa-volume-xmark text-gray-500";

                el.audioStatusText.textContent = "Sound Off";

            }

        }


        function checkTrialStatus() {

            if (APP_STATE.isSubscribed) {

                el.trialDaysText.textContent = "PRO Subscribed";

                return true;

            }


            const elapsedDays = Math.floor((Date.now() - parseInt(APP_STATE.trialStartDate)) / (1000 * 60 * 60 * 24));

            const daysRemaining = Math.max(0, 7 - elapsedDays);


            if (daysRemaining > 0) {

                el.trialDaysText.textContent = `${daysRemaining} Days Free Trial`;

                return true;

            } else {

                el.trialDaysText.textContent = `Trial Expired`;

                showPaywall();

                return false;

            }

        }

            el.modalPaywall.classList.remove('hidden');

        }

            el.viewDashboard.classList.remove('hidden');

            renderDashboard();

        });

        });

                    b.classList.add('bg-kindle-surface', 'text-kindle-cream');

                e.target.classList.add('bg-kindle-amber', 'text-black', 'font-semibold');

        });

                APP_STATE.currentPage++;

                updateReaderContent();

        });


        el.btnBookmark.addEventListener('click', () => {

            if (APP_STATE.currentStory) {

                toggleBookmark(APP_STATE.currentStory.id);

            }

        });

        document.getElementById('close-settings').addEventListener('click', () => el.modalSettings.classList.add('hidden'));

                applySettings();

            });

        });

        });
